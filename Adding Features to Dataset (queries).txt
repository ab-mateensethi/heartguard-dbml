-- =============================================
-- STEP 3: Features Ki Values Fill Karo
-- =============================================
 
-- AgeGroup
UPDATE heart_data
SET AgeGroup = CASE
    WHEN Age < 40 THEN 'Young'
    WHEN Age < 55 THEN 'Middle-Aged'
    ELSE 'Senior'
END;
 
-- BP_Category
UPDATE heart_data
SET BP_Category = CASE
    WHEN RestingBP < 120 THEN 'Normal'
    WHEN RestingBP < 130 THEN 'Elevated'
    WHEN RestingBP < 140 THEN 'High-Stage1'
    ELSE 'High-Stage2'
END;
 
-- Cholesterol_Category
UPDATE heart_data
SET Cholesterol_Category = CASE
    WHEN Cholesterol = 0   THEN 'Unknown'
    WHEN Cholesterol < 200 THEN 'Healthy'
    WHEN Cholesterol < 240 THEN 'Borderline'
    ELSE 'High'
END;
 
-- MaxHR_Category
UPDATE heart_data
SET MaxHR_Category = CASE
    WHEN MaxHR < 100 THEN 'Low'
    WHEN MaxHR < 150 THEN 'Moderate'
    ELSE 'High'
END;
 
-- RiskScore
UPDATE heart_data
SET RiskScore =
    (CASE WHEN Age > 55 THEN 1 ELSE 0 END) +
    (CASE WHEN RestingBP >= 140 THEN 1 ELSE 0 END) +
    (CASE WHEN Cholesterol > 240 THEN 1 ELSE 0 END) +
    (CASE WHEN FastingBS = 1 THEN 1 ELSE 0 END) +
    (CASE WHEN ExerciseAngina = 'Y' THEN 1 ELSE 0 END);
 
-- RiskLevel
UPDATE heart_data
SET RiskLevel = CASE
    WHEN RiskScore <= 1 THEN 'Low'
    WHEN RiskScore <= 3 THEN 'Medium'
    ELSE 'High'
END;
